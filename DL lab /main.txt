from tensorflow.keras.layers import Flatten, Dense

(x_train, y_train), (x_test, y_test) = cifar10.load_data()

x_train = x_train / 255.0
x_test = x_test / 255.0

y_train = to_categorical(y_train, 10)
y_test = to_categorical(y_test, 10)

class_names = [
    "Airplane",
    "Automobile",
    "Bird",
    "Cat",
    "Deer",
    "Dog",
    "Frog",
    "Horse",
    "Ship",
    "Truck"
]

def create_model(hidden_units, activation):
    model = Sequential([
        Flatten(input_shape=(32, 32, 3)),
        Dense(hidden_units[0], activation=activation),
        Dense(hidden_units[1], activation=activation),
        Dense(hidden_units[2], activation=activation),
        Dense(10, activation="softmax")
    ])

    model.compile(
        optimizer="adam",
        loss="categorical_crossentropy",
        metrics=["accuracy"]
    )

    return model

import time
import pandas as pd
import tensorflow as tf

from tensorflow.keras.datasets import cifar10
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Input, Flatten, Dense, Dropout
from tensorflow.keras.regularizers import l2
from tensorflow.keras.utils import to_categorical

# Reproducibility
tf.keras.utils.set_random_seed(42)

# 1. Load and prepare CIFAR-10 data
(x_train, y_train), (x_test, y_test) = cifar10.load_data()

x_train = x_train.astype("float32") / 255.0
x_test = x_test.astype("float32") / 255.0

y_train = to_categorical(y_train, num_classes=10)
y_test = to_categorical(y_test, num_classes=10)


# 2. Build the model
def build_model(init, dropout=False, reg=None):
    model = Sequential([
        Input(shape=(32, 32, 3)),
        Flatten()
    ])

    for units in [512, 256, 64]:
        model.add(Dense(
            units,
            activation="relu",
            kernel_initializer=init,
            kernel_regularizer=reg
        ))

        if dropout:
            model.add(Dropout(0.5))

    model.add(Dense(
        10,
        activation="softmax",
        kernel_initializer=init
    ))

    return model


# 3. Define experiments
experiments = [
    ("Baseline (Glorot Uniform)", "glorot_uniform", False, None),
    ("Xavier Normal", "glorot_normal", False, None),
    ("Kaiming (He Normal)", "he_normal", False, None),
    ("Dropout", "glorot_uniform", True, None),
    ("L2 Regularization", "glorot_uniform", False, l2(0.001)),
]

results = []

# 4. Train and evaluate models
for name, init, dropout, reg in experiments:
    for opt in ["Adam", "SGD"]:

        print(f"\nTraining: {name} | Optimizer: {opt}")

        model = build_model(init, dropout, reg)

        model.compile(
            optimizer=opt.lower(),
            loss="categorical_crossentropy",
            metrics=["accuracy"]
        )

        start = time.time()

        model.fit(
            x_train,
            y_train,
            epochs=5,
            batch_size=64,
            validation_split=0.1,
            verbose=1
        )

        train_time = time.time() - start

        # Evaluate only on the held-out test set
        loss, acc = model.evaluate(
            x_test, y_test, verbose=0
        )

        results.append([
            name, opt, loss, acc * 100, train_time
        ])

        del model


# 5. Display comparison
results_df = pd.DataFrame(results, columns=[
    "Model",
    "Optimizer",
    "Test Loss",
    "Test Accuracy (%)",
    "Training Time (s)"
])

print("\n" + "=" * 100)
print("FINAL COMPARISON")
print("=" * 100)
print(results_df.to_string(index=False))

print("\nBest model by test accuracy:")
print(
    results_df.loc[results_df["Test Accuracy (%)"].idxmax()]
)

max_length = 200

(x_train, y_train), (x_test, y_test) = imdb.load_data(
    num_words=vocab_size
)

print("Training samples:", len(x_train))
print("Testing samples:", len(x_test))

# 2. Pad sequences to equal length
x_train = pad_sequences(
    x_train, maxlen=max_length, padding="post", truncating="post"

    x_test, maxlen=max_length, padding="post", truncating="post"
)

    Embedding(vocab_size, 32),

    Dense(1, activation="sigmoid")

# 4. Compile the model
model.compile(
    optimizer="adam",

model.summary()

word_index = imdb.get_word_index()

def preprocess_review(review):
    words = review.lower().split()
    sequence = [1]  # IMDb sequence start token

    for word in words:
        # Map unknown words to the unknown token
        index = word_index.get(word, 2) + 3

        if index >= vocab_size:
            index = 2

        sequence.append(index)

    sequence = pad_sequences(
        [sequence],
        maxlen=max_length,
        padding="post",
        truncating="post"
    )

    return sequence

Experiment 6: MNIST Classification using a Vision CNN,


import time
import tensorflow as tf
import matplotlib.pyplot as plt
import numpy as np

from tensorflow.keras.datasets import mnist
from tensorflow.keras.applications import MobileNetV2
from tensorflow.keras.applications.mobilenet_v2 import preprocess_input
from tensorflow.keras.layers import Input, GlobalAveragePooling2D, Dense, Dropout
from tensorflow.keras.models import Model

# --------------------------------------------------
# 1. Load and preprocess MNIST dataset
# --------------------------------------------------

(x_train, y_train), (x_test, y_test) = mnist.load_data()

# Split training data into training and validation sets
x_val = x_train[55000:]
y_val = y_train[55000:]

x_train = x_train[:55000]
y_train = y_train[:55000]

# Convert grayscale images to 3-channel images
def preprocess_image(image, label):
    image = tf.cast(image, tf.float32)
    image = image[..., tf.newaxis]
    image = tf.image.grayscale_to_rgb(image)
    image = tf.image.resize(image, (64, 64))
    image = preprocess_input(image)
    return image, label

def create_dataset(images, labels, training=False):
    dataset = tf.data.Dataset.from_tensor_slices((images, labels))
    if training:
        dataset = dataset.shuffle(10000)
    dataset = dataset.map(
        preprocess_image,
        num_parallel_calls=tf.data.AUTOTUNE
    )
    dataset = dataset.batch(128).prefetch(tf.data.AUTOTUNE)
    return dataset

train_ds = create_dataset(x_train, y_train, True)
val_ds = create_dataset(x_val, y_val)
test_ds = create_dataset(x_test, y_test)

# --------------------------------------------------
# 2. Load pretrained CNN model
# --------------------------------------------------

base_model = MobileNetV2(
    weights="imagenet",
    include_top=False,
    input_shape=(64, 64, 3)
)

# Build classification model
inputs = Input(shape=(64, 64, 3))
x = base_model(inputs, training=False)
x = GlobalAveragePooling2D()(x)
x = Dropout(0.2)(x)
outputs = Dense(10, activation="softmax")(x)

model = Model(inputs, outputs)

# --------------------------------------------------
# 3. Fixed Feature Extraction
# --------------------------------------------------

base_model.trainable = False

model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=0.001),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"]
)

start_time = time.time()

history_fixed = model.fit(
    train_ds,
    validation_data=val_ds,
    epochs=4
)

fixed_time = time.time() - start_time

fixed_loss, fixed_accuracy = model.evaluate(test_ds, verbose=0)

# --------------------------------------------------
# 4. Fine-Tuning
# --------------------------------------------------

# Unfreeze the last layers of the pretrained CNN
base_model.trainable = True

for layer in base_model.layers[:-30]:
    layer.trainable = False

# Keep batch normalization layers frozen for stability
for layer in base_model.layers:
    if isinstance(layer, tf.keras.layers.BatchNormalization):
        layer.trainable = False

model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=0.00001),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"]
)

start_time = time.time()

history_fine = model.fit(
    train_ds,
    validation_data=val_ds,
    epochs=4
)

fine_time = time.time() - start_time

fine_loss, fine_accuracy = model.evaluate(test_ds, verbose=0)

# --------------------------------------------------
# 5. Display Results
# --------------------------------------------------

print("\n========== FINAL RESULTS ==========")

print(f"Fixed Feature Extraction Accuracy: {fixed_accuracy * 100:.2f}%")
print(f"Fixed Feature Extraction Loss:     {fixed_loss:.4f}")
print(f"Fixed Feature Extraction Time:     {fixed_time:.2f} seconds")

print(f"\nFine-Tuning Accuracy:              {fine_accuracy * 100:.2f}%")
print(f"Fine-Tuning Loss:                  {fine_loss:.4f}")
print(f"Fine-Tuning Time:                  {fine_time:.2f} seconds")

print("\n========== PERFORMANCE COMPARISON ==========")

print(f"Accuracy Improvement: {(fine_accuracy - fixed_accuracy) * 100:.2f}%")
print(f"Loss Difference:      {fixed_loss - fine_loss:.4f}")

# --------------------------------------------------
