    model = Sequential([
        Flatten(input_shape=(32, 32, 3)),

        Dense(hidden_units[1], activation=activation),
        Dense(hidden_units[2], activation=activation),

import time
import pandas as pd
import tensorflow as tf

from tensorflow.keras.datasets import cifar10
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Input, Flatten, Dense, Dropout
from tensorflow.keras.regularizers import l2
from tensorflow.keras.utils import to_categorical

# Reproducibility
tf.keras.utils.set_random_seed(42)

# 1. Load and prepare CIFAR-10 data
(x_train, y_train), (x_test, y_test) = cifar10.load_data()

x_train = x_train.astype("float32") / 255.0
x_test = x_test.astype("float32") / 255.0

y_train = to_categorical(y_train, num_classes=10)
y_test = to_categorical(y_test, num_classes=10)


# 2. Build the model
def build_model(init, dropout=False, reg=None):
    model = Sequential([
        Input(shape=(32, 32, 3)),
        Flatten()
    ])

    for units in [512, 256, 64]:
        model.add(Dense(
            units,
            activation="relu",
            kernel_initializer=init,
            kernel_regularizer=reg
        ))

        if dropout:
            model.add(Dropout(0.5))

    model.add(Dense(
        10,
        activation="softmax",
        kernel_initializer=init
    ))

    return model


# 3. Define experiments
experiments = [
    ("Baseline (Glorot Uniform)", "glorot_uniform", False, None),
    ("Xavier Normal", "glorot_normal", False, None),
    ("Kaiming (He Normal)", "he_normal", False, None),
    ("Dropout", "glorot_uniform", True, None),
    ("L2 Regularization", "glorot_uniform", False, l2(0.001)),
]

results = []

# 4. Train and evaluate models
for name, init, dropout, reg in experiments:
    for opt in ["Adam", "SGD"]:

        print(f"\nTraining: {name} | Optimizer: {opt}")

        model = build_model(init, dropout, reg)

        model.compile(
            optimizer=opt.lower(),
            loss="categorical_crossentropy",
            metrics=["accuracy"]
        )

        start = time.time()

        model.fit(
            x_train,
            y_train,
            epochs=5,
            batch_size=64,
            validation_split=0.1,
            verbose=1
        )

        train_time = time.time() - start

        # Evaluate only on the held-out test set
        loss, acc = model.evaluate(
            x_test, y_test, verbose=0
        )

        results.append([
            name, opt, loss, acc * 100, train_time
        ])

        del model


# 5. Display comparison
results_df = pd.DataFrame(results, columns=[
    "Model",
    "Optimizer",
    "Test Loss",
    "Test Accuracy (%)",
    "Training Time (s)"
])

print("\n" + "=" * 100)
print("FINAL COMPARISON")
print("=" * 100)
print(results_df.to_string(index=False))

print("\nBest model by test accuracy:")
print(
    results_df.loc[results_df["Test Accuracy (%)"].idxmax()]
)

import numpy as np
import matplotlib.pyplot as plt
import tensorflow as tf

from tensorflow.keras.datasets import mnist
from tensorflow.keras.models import Sequential, Model
from tensorflow.keras.layers import Dense, Flatten
from tensorflow.keras.optimizers import Adam
from tensorflow.keras.utils import to_categorical
from tensorflow.keras.applications import VGG19

# 1. Load MNIST dataset
(x_train, y_train), (x_test, y_test) = mnist.load_data()

x_train, y_train = x_train[:5000], y_train[:5000]
x_test, y_test = x_test[:500], y_test[:500]

# 2. Normalize data
x_train = x_train / 255.0
x_test = x_test / 255.0

y_train = to_categorical(y_train, 10)
y_test = to_categorical(y_test, 10)

# 3. Fixed Feature Extractor Model
model_frozen = Sequential([
    Flatten(input_shape=(28, 28)),
    Dense(256, activation="relu"),
    Dense(10, activation="softmax")
])

model_frozen.compile(
    optimizer=Adam(learning_rate=0.001),
    loss="categorical_crossentropy",
    metrics=["accuracy"]
)

history_frozen = model_frozen.fit(
    x_train, y_train,
    epochs=5,
    batch_size=32,
    validation_data=(x_test, y_test)
)

# 4. Convert grayscale images to RGB
x_train_rgb = tf.image.grayscale_to_rgb(
    tf.expand_dims(x_train, axis=-1)
)

x_test_rgb = tf.image.grayscale_to_rgb(
    tf.expand_dims(x_test, axis=-1)
)

# 5. Resize images to 224x224
x_train_rgb = tf.image.resize(x_train_rgb, (224, 224))
x_test_rgb = tf.image.resize(x_test_rgb, (224, 224))

# 6. Load pretrained VGG19 model
base_model = VGG19(
    weights="imagenet",
    include_top=False,
    input_shape=(224, 224, 3)
)

# Freeze VGG19 layers
for layer in base_model.layers:
    layer.trainable = False

# 7. Add classification layers
x = Flatten()(base_model.output)
x = Dense(256, activation="relu")(x)
output = Dense(10, activation="softmax")(x)

model_vgg = Model(
    inputs=base_model.input,
    outputs=output
)

model_vgg.compile(
    optimizer=Adam(learning_rate=0.001),
    loss="categorical_crossentropy",
    metrics=["accuracy"]
)

# 8. Train VGG19 model
history_vgg = model_vgg.fit(
    x_train_rgb,
    y_train,
    epochs=5,
    batch_size=64,
    validation_data=(x_test_rgb, y_test)
)

# 9. Evaluate model
loss, accuracy = model_vgg.evaluate(x_test_rgb, y_test)

print("VGG19 Test Accuracy:", accuracy * 100, "%")
print("VGG19 Test Loss:", loss)

# 10. Plot accuracy comparison
plt.plot(history_frozen.history["accuracy"], label="Frozen Model")
plt.plot(history_frozen.history["val_accuracy"], label="Frozen Validation")

plt.plot(history_vgg.history["accuracy"], label="VGG19 Model")
plt.plot(history_vgg.history["val_accuracy"], label="VGG19 Validation")

plt.xlabel("Epoch")
plt.ylabel("Accuracy")
plt.title("Model Accuracy Comparison")
plt.legend()
plt.show()

Exp 9  -> IMDB Sentiment Analysis Using Simple RNN, LSTM and GRU


import tensorflow as tf
import pandas as pd
import time

from tensorflow.keras.datasets import imdb
from tensorflow.keras.preprocessing.sequence import pad_sequences
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Embedding, SimpleRNN, LSTM, GRU, Dense

# 1. Load IMDB dataset
vocab_size = 10000
max_length = 200

(x_train, y_train), (x_test, y_test) = imdb.load_data(num_words=vocab_size)

print("Training samples:", len(x_train))
print("Testing samples:", len(x_test))

# 2. Padding
x_train = pad_sequences(x_train, maxlen=max_length, padding="post")
x_test = pad_sequences(x_test, maxlen=max_length, padding="post")

print("Training shape:", x_train.shape)
print("Testing shape:", x_test.shape)

# 3. Build model
def build_model(model_type):
    model = Sequential([
        tf.keras.Input(shape=(max_length,)),
        Embedding(vocab_size, 64)
    ])

    if model_type == "Simple RNN":
        model.add(SimpleRNN(64))
    elif model_type == "LSTM":
        model.add(LSTM(64))
    else:
        model.add(GRU(64))

    model.add(Dense(1, activation="sigmoid"))

    model.compile(
        optimizer="adam",
        loss="binary_crossentropy",
        metrics=["accuracy"]
    )

    return model

# 4. Compare models
architectures = ["Simple RNN", "LSTM", "GRU"]
results = []

for name in architectures:
    print("\nTraining:", name)

    model = build_model(name)

    start = time.time()

    model.fit(
        x_train, y_train,
        epochs=6,
        batch_size=64,
        validation_split=0.2,
        verbose=1
    )

    training_time = time.time() - start

    # 5. Evaluate model
    loss, accuracy = model.evaluate(x_test, y_test, verbose=0)

    results.append([
        name,
        accuracy * 100,
        loss,
        training_time
    ])

# 6. Display results
df = pd.DataFrame(results, columns=[
    "Model",
    "Test Accuracy (%)",
    "Test Loss",
    "Training Time (s)"
])

print("\nMODEL COMPARISON")
print("=" * 70)
print(df.to_string(index=False))

# 7. Final analysis
print("\nBest Accuracy:", df.loc[df["Test Accuracy (%)"].idxmax(), "Model"])
print("Lowest Loss:", df.loc[df["Test Loss"].idxmin(), "Model"])
